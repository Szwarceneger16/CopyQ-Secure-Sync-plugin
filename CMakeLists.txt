cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" CQS_RELEASE_VERSION)
string(STRIP "${CQS_RELEASE_VERSION}" CQS_RELEASE_VERSION)
if(NOT CQS_RELEASE_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)(-rc\\.[1-9][0-9]*)?$")
  message(FATAL_ERROR "VERSION must contain a release version or numbered release candidate")
endif()
project(CopyQSecureSync VERSION "${CMAKE_MATCH_1}" LANGUAGES CXX)
configure_file(native/version.h.in "${CMAKE_BINARY_DIR}/cqs-version.h" @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(CQS_BUILD_NATIVE_PLUGIN "Build the CopyQ 17 native plugin with opt-in standalone storage" ON)
option(CQS_BUILD_STANDALONE_LAB "Build the opt-in standalone storage laboratory module" OFF)
option(CQS_TEST_LEGACY_HOST_BRIDGE "Test the superseded optional patched-host bridge" OFF)
option(CQS_BUILD_UI_TEST_PLUGIN "Build the public-Qt GUI driver for disposable process tests only" OFF)

find_package(ZLIB REQUIRED)
set(CQS_CORE_SOURCES src/compression.cpp src/history.cpp src/events.cpp src/import_transaction.cpp src/state_record.cpp src/packed_items.cpp)
if(WIN32)
  find_package(Threads REQUIRED)
  list(APPEND CQS_CORE_SOURCES src/crypto_windows.cpp src/state_store_windows.cpp src/storage_windows.cpp native/key_reader_windows.cpp)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  find_package(Threads REQUIRED)
  list(APPEND CQS_CORE_SOURCES src/crypto.cpp src/storage_linux.cpp native/key_reader_linux.cpp src/state_store_linux.cpp)
else()
  message(FATAL_ERROR "No audited platform backend for this target")
endif()
add_library(cqs-core STATIC ${CQS_CORE_SOURCES})
set_target_properties(cqs-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(cqs-core PUBLIC src native)
target_link_libraries(cqs-core PUBLIC ZLIB::ZLIB)
if(WIN32)
  target_compile_definitions(cqs-core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX)
  target_link_libraries(cqs-core PUBLIC bcrypt advapi32 Threads::Threads)
else()
  target_link_libraries(cqs-core PUBLIC OpenSSL::Crypto Threads::Threads)
endif()

if(CQS_BUILD_NATIVE_PLUGIN)
  # Linux distribution builds use their own Qt ABI. Windows path handling is
  # validated with Qt 6.10; do not lower that floor with the Linux SDK.
  if(WIN32)
    set(CQS_MIN_QT 6.10)
  else()
    set(CQS_MIN_QT 6.4)
  endif()
  find_package(Qt6 ${CQS_MIN_QT} REQUIRED COMPONENTS Core Gui Widgets)
  find_package(Git REQUIRED)
  set(COPYQ_SOURCE_DIR "" CACHE PATH "Clean CopyQ checkout at the pinned v17.0.0 commit")
  set(CQS_COPYQ_COMMIT d0c179f75c04974bfe801289ea6e58a1a49535fc)
  if(NOT EXISTS "${COPYQ_SOURCE_DIR}/src/item/itemwidget.h")
    message(FATAL_ERROR "Provide COPYQ_SOURCE_DIR; configuring never downloads upstream sources")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${COPYQ_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE COPYQ_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE COPYQ_GIT_ERROR)
  if(COPYQ_GIT_ERROR OR NOT COPYQ_COMMIT STREQUAL CQS_COPYQ_COMMIT)
    message(FATAL_ERROR "CopyQ source must be the reviewed v17.0.0 commit ${CQS_COPYQ_COMMIT}")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${COPYQ_SOURCE_DIR}" status --porcelain -- src
    OUTPUT_VARIABLE COPYQ_DIRTY RESULT_VARIABLE COPYQ_GIT_ERROR)
  if(COPYQ_GIT_ERROR OR NOT COPYQ_DIRTY STREQUAL "")
    message(FATAL_ERROR "CopyQ API support sources are modified; use a clean pinned checkout")
  endif()
  file(STRINGS "${COPYQ_SOURCE_DIR}/src/item/itemwidget.h" COPYQ_IID_LINE
    REGEX "^#define COPYQ_PLUGIN_ITEM_LOADER_ID ")
  if(NOT COPYQ_IID_LINE STREQUAL "#define COPYQ_PLUGIN_ITEM_LOADER_ID \"com.github.hluk.copyq.itemloader/17.0.0\"")
    message(FATAL_ERROR "Unexpected CopyQ plugin ABI")
  endif()
  # Same interface support sources used by CopyQ's own plugin build macro.
  # We do not link to itemsync or any other installed plugin, nor build CopyQ.
  add_library(cqs-copyq-api STATIC
    "${COPYQ_SOURCE_DIR}/src/item/itemwidget.cpp"
    "${COPYQ_SOURCE_DIR}/src/common/mimetypes.cpp")
  target_include_directories(cqs-copyq-api PUBLIC "${COPYQ_SOURCE_DIR}/src")
  target_link_libraries(cqs-copyq-api PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets)
  set_target_properties(cqs-copyq-api PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(cqs-settings STATIC native/profile.cpp native/settings_widget.cpp)
  target_include_directories(cqs-settings PUBLIC native)
  target_link_libraries(cqs-settings PUBLIC Qt6::Core Qt6::Widgets)
  set_target_properties(cqs-settings PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
  add_library(cqs-model-clipboard STATIC native/model_clipboard.cpp native/gui_clipboard.cpp native/row_protection.cpp native/native_blob.cpp)
  if(WIN32)
    target_sources(cqs-model-clipboard PRIVATE native/native_blob_windows.cpp)
  else()
    target_sources(cqs-model-clipboard PRIVATE native/native_blob_linux.cpp)
  endif()
  target_include_directories(cqs-model-clipboard PUBLIC native)
  target_link_libraries(cqs-model-clipboard PUBLIC cqs-core cqs-copyq-api Qt6::Core)
  set_target_properties(cqs-model-clipboard PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(cqs-standalone-storage STATIC native/standalone_storage.cpp native/standalone_storage_store.cpp native/key_vault.cpp native/unified_state.cpp native/seed_history.cpp native/history_transition.cpp native/transport_watch.cpp)
  target_link_libraries(cqs-standalone-storage PUBLIC cqs-model-clipboard cqs-settings)
  set_target_properties(cqs-standalone-storage PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
  add_library(securesync MODULE native/plugin.cpp native/inspector.cpp)
  file(WRITE "${CMAKE_BINARY_DIR}/cqs-plugin.json" "{\"Version\":\"${CQS_RELEASE_VERSION}\",\"Product\":\"CopyQ Secure Sync\"}\n")
  target_include_directories(securesync PRIVATE "${CMAKE_BINARY_DIR}")
  target_link_libraries(securesync PRIVATE cqs-settings cqs-copyq-api cqs-core cqs-standalone-storage)
  set_target_properties(securesync PROPERTIES AUTOMOC ON)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/cqs-abi.json" CONTENT
    "{\"schema\":1,\"copyq\":\"17.0.0\",\"qt\":\"${Qt6Core_VERSION}\",\"processor\":\"${CMAKE_SYSTEM_PROCESSOR}\",\"pointer_size\":${CMAKE_SIZEOF_VOID_P},\"compiler\":\"${CMAKE_CXX_COMPILER_ID}\"}\n")
  if(UNIX)
    target_link_options(securesync PRIVATE "-Wl,--no-undefined")
  endif()
  # Staging only. Installing this project never edits the actual CopyQ profile.
  install(TARGETS securesync LIBRARY DESTINATION copyq/plugins RUNTIME DESTINATION copyq/plugins)
  if(CQS_BUILD_STANDALONE_LAB)
    add_library(securesynclab MODULE native/standalone_lab.cpp native/standalone_lab_store.cpp)
    target_link_libraries(securesynclab PRIVATE cqs-model-clipboard)
    set_target_properties(securesynclab PROPERTIES AUTOMOC ON)
    if(UNIX)
      target_link_options(securesynclab PRIVATE "-Wl,--no-undefined")
    endif()
    # A separate experimental install component; excluded from normal preview ZIPs.
    install(TARGETS securesynclab LIBRARY DESTINATION copyq/plugins RUNTIME DESTINATION copyq/plugins COMPONENT StandaloneLab)
  endif()
  if(BUILD_TESTING)
    add_executable(cqs-history-transition-test tests/history_transition_test.cpp native/history_transition.cpp)
    target_compile_definitions(cqs-history-transition-test PRIVATE CQS_TRANSITION_TESTING)
    target_link_libraries(cqs-history-transition-test PRIVATE cqs-standalone-storage)

    set_tests_properties(native-history-transition PROPERTIES TIMEOUT 360)
    if(CQS_BUILD_UI_TEST_PLUGIN)
      # Public Qt ABI only: QtTest uses private symbols tied to a minor release.
      add_library(cqs-ui-driver MODULE tests/ui_driver.cpp)
      target_link_libraries(cqs-ui-driver PRIVATE cqs-copyq-api)
      set_target_properties(cqs-ui-driver PROPERTIES AUTOMOC ON)
      if(UNIX)
        target_link_options(cqs-ui-driver PRIVATE "-Wl,--no-undefined")
      endif()
      # Intentionally not installed or included in development/release packages.
    endif()
    add_executable(cqs-key-vault-test tests/key_vault_test.cpp native/key_vault.cpp)
    if(WIN32)
      set(CQS_TEST_STATE_BACKEND src/state_store_windows.cpp)
      set(CQS_TEST_STORAGE_BACKEND src/storage_windows.cpp)
    else()
      set(CQS_TEST_STATE_BACKEND src/state_store_linux.cpp)
      set(CQS_TEST_STORAGE_BACKEND src/storage_linux.cpp)
    endif()
    add_executable(cqs-unified-state-test tests/unified_state_test.cpp ${CQS_TEST_STATE_BACKEND} ${CQS_TEST_STORAGE_BACKEND})
    target_compile_definitions(cqs-unified-state-test PRIVATE CQS_STATE_TESTING CQS_STORAGE_TESTING)
    target_link_libraries(cqs-unified-state-test PRIVATE cqs-standalone-storage)
    add_test(NAME native-unified-state COMMAND cqs-unified-state-test)
    set_tests_properties(native-unified-state PROPERTIES TIMEOUT 180)
    target_compile_definitions(cqs-key-vault-test PRIVATE CQS_VAULT_TESTING)
    target_link_libraries(cqs-key-vault-test PRIVATE cqs-standalone-storage)
    add_test(NAME native-key-vault COMMAND cqs-key-vault-test)
    set_tests_properties(native-key-vault PROPERTIES TIMEOUT 120)
    add_executable(cqs-standalone-storage-test tests/standalone_storage_test.cpp
      "${COPYQ_SOURCE_DIR}/src/item/clipboardmodel.cpp"
      "${COPYQ_SOURCE_DIR}/src/item/clipboarditem.cpp"
      "${COPYQ_SOURCE_DIR}/src/item/serialize.cpp"
      "${COPYQ_SOURCE_DIR}/src/common/encryption.cpp"
      "${COPYQ_SOURCE_DIR}/src/common/textdata.cpp")
    target_link_libraries(cqs-standalone-storage-test PRIVATE cqs-standalone-storage)
    set_target_properties(cqs-standalone-storage-test PROPERTIES AUTOMOC ON)
    add_test(NAME native-standalone-storage COMMAND cqs-standalone-storage-test "$<TARGET_FILE:securesync>")
    set_tests_properties(native-standalone-storage PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    add_executable(cqs-settings-test tests/native_settings_test.cpp)
    target_link_libraries(cqs-settings-test PRIVATE cqs-settings)
    add_test(NAME native-settings COMMAND cqs-settings-test)
    set_tests_properties(native-settings PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    add_executable(cqs-loader-test tests/native_loader_test.cpp)
    target_include_directories(cqs-loader-test PRIVATE "${CMAKE_BINARY_DIR}")
    target_link_libraries(cqs-loader-test PRIVATE cqs-copyq-api Qt6::Widgets)
    add_test(NAME native-loader COMMAND cqs-loader-test "$<TARGET_FILE:securesync>")
    set_tests_properties(native-loader PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    add_executable(cqs-model-clipboard-test tests/native_model_clipboard_test.cpp
      "${COPYQ_SOURCE_DIR}/src/item/clipboardmodel.cpp"
      "${COPYQ_SOURCE_DIR}/src/item/clipboarditem.cpp"
      "${COPYQ_SOURCE_DIR}/src/item/serialize.cpp"
      "${COPYQ_SOURCE_DIR}/src/common/encryption.cpp"
      "${COPYQ_SOURCE_DIR}/src/common/textdata.cpp")
    target_link_libraries(cqs-model-clipboard-test PRIVATE cqs-model-clipboard)
    add_test(NAME native-model-clipboard COMMAND cqs-model-clipboard-test)
    set_tests_properties(native-model-clipboard PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
    add_executable(cqs-inspector-test tests/native_inspector_test.cpp native/inspector.cpp)
    target_link_libraries(cqs-inspector-test PRIVATE cqs-settings cqs-core)
    set_target_properties(cqs-inspector-test PROPERTIES AUTOMOC ON)
    add_test(NAME native-inspector COMMAND cqs-inspector-test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    set_tests_properties(native-inspector PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
    if(CQS_BUILD_STANDALONE_LAB)
      add_executable(cqs-standalone-lab-test tests/standalone_lab_test.cpp
        "${COPYQ_SOURCE_DIR}/src/item/clipboardmodel.cpp"
        "${COPYQ_SOURCE_DIR}/src/item/clipboarditem.cpp"
        "${COPYQ_SOURCE_DIR}/src/common/textdata.cpp")
      target_link_libraries(cqs-standalone-lab-test PRIVATE cqs-model-clipboard)
      add_test(NAME native-standalone-lab COMMAND cqs-standalone-lab-test "$<TARGET_FILE:securesynclab>")
      set_tests_properties(native-standalone-lab PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
    endif()
    if(CQS_TEST_LEGACY_HOST_BRIDGE)
      # Historical adapter is only linked into this regression executable.
      # Actual QCA-enabled CopyQ encryption, not the no-QCA model-test stub.
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      find_package(Qca-qt6 2.3.10 REQUIRED)
      set(CQS_HOST_EXPORT "${CMAKE_CURRENT_BINARY_DIR}/copyq-host")
      execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/host/prepare.py"
        --copyq "${COPYQ_SOURCE_DIR}" --out "${CQS_HOST_EXPORT}"
        RESULT_VARIABLE CQS_HOST_EXPORT_ERROR)
      if(CQS_HOST_EXPORT_ERROR)
        message(FATAL_ERROR "CopyQ host adaptation did not apply cleanly")
      endif()
      add_executable(cqs-host-persistence-test tests/native_host_persistence_test.cpp
        native/host_clipboard.cpp
        host/tab_persistence.cpp
        "${CQS_HOST_EXPORT}/src/item/serialize.cpp"
        "${CQS_HOST_EXPORT}/src/item/clipboardmodel.cpp"
        "${CQS_HOST_EXPORT}/src/item/clipboarditem.cpp"
        "${CQS_HOST_EXPORT}/src/common/encryption.cpp"
        "${CQS_HOST_EXPORT}/src/common/textdata.cpp")
      target_include_directories(cqs-host-persistence-test BEFORE PRIVATE "${CQS_HOST_EXPORT}/src")
      target_include_directories(cqs-host-persistence-test PRIVATE host)
      target_compile_definitions(cqs-host-persistence-test PRIVATE WITH_QCA_ENCRYPTION CQS_HOST_TESTING)
      target_link_libraries(cqs-host-persistence-test PRIVATE cqs-model-clipboard qca-qt6)
      set_target_properties(cqs-host-persistence-test PROPERTIES AUTOMOC ON)
      add_test(NAME native-host-persistence COMMAND cqs-host-persistence-test "$<TARGET_FILE:securesync>")
      set_tests_properties(native-host-persistence PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
      # Compile the actual GUI/factory adaptations as well, so an integration patch
      # which merely applies but does not compile cannot pass the native gate.
      add_library(cqs-host-integration-check OBJECT
        "${CQS_HOST_EXPORT}/src/gui/clipboardbrowser.cpp"
        "${CQS_HOST_EXPORT}/src/gui/clipboardbrowserplaceholder.cpp"
        "${CQS_HOST_EXPORT}/src/item/itemfactory.cpp"
        "${CQS_HOST_EXPORT}/src/item/itemstore.cpp")
      target_include_directories(cqs-host-integration-check PRIVATE "${CQS_HOST_EXPORT}/src")
      target_compile_definitions(cqs-host-integration-check PRIVATE WITH_QCA_ENCRYPTION)
      target_link_libraries(cqs-host-integration-check PRIVATE Qt6::Widgets qca-qt6)
    endif()
  endif()
endif()
